// Today's reader, copied verbatim from tasks.ts on the integration branch (parseEvidenceRefs).
function oldRefs(value: string): string[] {
  return value
    .trim()
    .replace(/^\((.*)\)$/, "$1")
    .split(",")
    .map((s) => s.trim().replace(/^`|`$/g, ""))
    .filter((s) => s !== "" && !/\s/.test(s) && (s.includes("/") || /\.\w+$/.test(s)));
}
function oldCriteria(section: string): { pass: boolean | null; evidence: string[] }[] {
  const out: { pass: boolean | null; evidence: string[] }[] = [];
  for (const raw of section.split("\n")) {
    const line = raw.trim();
    const box = /^-\s+\[([ xX])\]\s+(.+)$/.exec(line);
    if (box) { if (!/^\(.*\)$/.test((box[2] ?? "").trim())) out.push({ pass: box[1] !== " ", evidence: [] }); continue; }
    const bullet = /^-\s+(\S.*)$/.exec(line);
    if (bullet) { if (!/^\(.*\)$/.test((bullet[1] ?? "").trim())) out.push({ pass: null, evidence: [] }); continue; }
    const ev = /^evidence:\s*(.*)$/i.exec(line);
    const last = out[out.length - 1];
    if (ev && last) last.evidence.push(...oldRefs(ev[1] ?? ""));
  }
  return out;
}
function section(body: string, name: string): string {
  const lines: string[] = [];
  let on = false;
  for (const line of body.split("\n")) {
    const h = /^##\s+(.+?)\s*$/.exec(line);
    if (h) { on = (h[1] ?? "").trim() === name; continue; }
    if (on) lines.push(line);
  }
  return lines.join("\n");
}
