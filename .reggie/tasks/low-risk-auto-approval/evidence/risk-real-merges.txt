// Read-only probe over this repository's landed loop merges (AC26): the three risk classes the
// policy report computes, for each task as it stood when it was merged. It runs only `git log`,
// `git rev-parse`, `git merge-base`, `git ls-tree`, `git cat-file` and `git diff --name-only` against
// the repository, writes nothing there, and refuses to start unless its own working directory is
// under the OS temp directory (AC50).
//
// For each landing merge: the base is the merge's first parent and the branch tip its second. The
// config (the risk rules) and the plan are read from the base, as the report reads them; when the
// plan was not on the base the branch's copy is used and the row says so.
import path from "node:path";
const here = process.cwd();
if (!/^\/(private\/)?(tmp|var\/folders)\//.test(here)) {
  console.error(`refusing to run in ${here}: not under the OS temp directory`);
  process.exit(98);
}
console.log(`running in ${here}`);
const SRC = process.env.REGGIE_SRC ?? "";
const ROOT = process.env.REGGIE_READ_ROOT ?? "";
if (!SRC || !ROOT) throw new Error("set REGGIE_SRC (this branch's packages/reggie/src) and REGGIE_READ_ROOT (the repository to read)");
const { git, fileAtCommit, mergeBase, changedPathsNoRenames } = await import(path.join(SRC, "git.ts"));
const { parseConfig } = await import(path.join(SRC, "people.ts"));
const { parsePlan, riskFromFiles } = await import(path.join(SRC, "plan.ts"));
const { controlFileReason } = await import(path.join(SRC, "policy.ts"));

const RANK: Record<string, number> = { low: 0, medium: 1, high: 2, unset: 2 };
const NAMES = ["low", "medium", "high"];
const covers = (entry: string, file: string): boolean => entry === file || (entry.endsWith("/") && file.startsWith(entry));
const sha = (rev: string): string | null => {
  const out = git(["rev-parse", "--verify", "--quiet", `${rev}^{commit}`], { cwd: ROOT, allowFailure: true }).stdout.trim();
  return /^[0-9a-f]{40}$/.test(out) ? out : null;
};

const branch = process.env.REGGIE_READ_BRANCH ?? "repo-manager";
const merges = git(["log", "--first-parent", "--merges", "--format=%H%x1f%s", branch], { cwd: ROOT }).stdout.trim().split("\n");
const rows: Record<string, unknown>[] = [];
const detail: string[] = [];
let unplannedCode = 0, raised = 0, control = 0;
for (const line of merges) {
  const [merge = "", subject = ""] = line.split("\x1f");
  const slug = /task\/([a-z0-9-]+)/.exec(subject)?.[1];
  if (!slug) continue;
  const base = sha(`${merge}^1`);
  const tip = sha(`${merge}^2`);
  if (!base || !tip) continue;
  const planOnBase = fileAtCommit(ROOT, base, `.reggie/tasks/${slug}/plan.md`);
  const planText = planOnBase ?? fileAtCommit(ROOT, tip, `.reggie/tasks/${slug}/plan.md`);
  if (planText === null) continue;
  const plan = parsePlan(planText);
  const rules = parseConfig(fileAtCommit(ROOT, base, ".reggie/config.yaml")).risk;
  const fork = mergeBase(ROOT, base, tip);
  const changedAll: string[] = fork ? changedPathsNoRenames(ROOT, fork, tip) ?? [] : [];
  const counted = changedAll.filter((f) => !f.startsWith(`.reggie/tasks/${slug}/`) && !f.startsWith(".reggie/journal/"));
  const planned = plan.files.map((f: string) => f.replace(/^\.\/+/, ""));
  const unplanned = counted.filter((f) => !planned.some((p: string) => covers(p, f)));
  const code = unplanned.filter((f) => !f.startsWith(".reggie/"));
  const planLine: string = plan.meta.risk;
  const planFiles: string = riskFromFiles(planned, rules);
  const changed: string = riskFromFiles(counted, rules);
  const effective = NAMES[Math.max(RANK[planLine] ?? 2, RANK[planFiles] ?? 2, RANK[changed] ?? 2)];
  const controls = changedAll.filter((f) => controlFileReason(slug, f) !== null);
  if (code.length > 0) unplannedCode += 1;
  if ((RANK[effective ?? "high"] ?? 2) > (RANK[planLine] ?? 2)) raised += 1;
  if (controls.length > 0) control += 1;
  rows.push({ slug: slug.slice(0, 34), planOnBase: planOnBase !== null, planLine, planFiles, changed, effective, counted: counted.length, unplanned: unplanned.length, unplannedCode: code.length, controlFiles: controls.length });
  detail.push(`${slug}\n   unplanned code: ${code.join(", ") || "none"}\n   unplanned records: ${unplanned.filter((f) => f.startsWith(".reggie/")).length} under .reggie/ (notes and the intake file, mostly)\n   control files changed: ${controls.join(", ") || "none"}`);
}
console.table(rows);
console.log(`landed merges read: ${rows.length}`);
console.log(`tasks with a code file the plan never named: ${unplannedCode}`);
console.log(`tasks whose effective class is higher than the plan's line: ${raised}`);
for (const r of rows) if ((RANK[String(r.effective)] ?? 2) > (RANK[String(r.planLine)] ?? 2)) console.log(`   raised: ${r.slug} from ${r.planLine} to ${r.effective} (${r.counted} files counted)`);
console.log(`tasks that changed a control file, and so would always go to a person: ${control}`);
console.log("");
for (const d of detail) console.log(d);
