// Read-only probe (AC38, AC2): the task list of one working tree, as two builds derive it. Run once
// with DIST pointing at a dist compiled from the base commit and once at this branch's own; both
// read the same tree (REGGIE_TREE), and nothing is written to it. With PLANS=1 it prints, instead,
// parsePlan(...).criteria for every plan under .reggie/tasks/ in that tree.
// Refuses to start unless its own working directory is under the OS temp directory (AC50).
import { createHash } from "node:crypto";
import { existsSync, readdirSync, readFileSync } from "node:fs";
import path from "node:path";
const here = process.cwd();
if (!/^\/(private\/)?(tmp|var\/folders)\//.test(here)) {
  console.error(`refusing to run in ${here}: not under the OS temp directory`);
  process.exit(98);
}
const DIST = process.env.DIST ?? "";
const TREE = process.env.REGGIE_TREE ?? "";
if (!DIST || !TREE) throw new Error("set DIST (a compiled dist/) and REGGIE_TREE (the working tree to read)");
const { repoPaths } = await import(path.join(DIST, "paths.js"));
const { loadConfig } = await import(path.join(DIST, "people.js"));
if (process.env.PLANS === "1") {
  const { parsePlan } = await import(path.join(DIST, "plan.js"));
  const tasks = path.join(TREE, ".reggie/tasks");
  for (const slug of readdirSync(tasks).sort()) {
    const file = path.join(tasks, slug, "plan.md");
    if (!existsSync(file)) continue;
    const criteria = parsePlan(readFileSync(file, "utf8")).criteria;
    console.log(`${slug}\t${criteria.length}\t${createHash("sha256").update(JSON.stringify(criteria)).digest("hex").slice(0, 16)}`);
  }
} else {
  const { listTasks } = await import(path.join(DIST, "tasks.js"));
  const paths = repoPaths(TREE);
  for (const t of listTasks(paths, loadConfig(paths), { includeDone: true })) console.log([t.slug, t.state, t.reason, t.risk].join("\t"));
}
